Query 1: Enumerate Registered Human Owners (Maps to CQ1 & CQ2)

PREFIX pet: <http://www.semanticweb.org/pet-management-ontology#> 
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> 
PREFIX owl: <http://www.w3.org/2002/07/owl#> 
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> 
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#> 
SELECT ?sahipID ?sahipIsmi ?telefon ?adres 
WHERE { 
    ?sahipURI rdf:type pet:Owner . 
    ?sahipURI pet:hasOwnerId ?sahipID . 
    ?sahipURI pet:hasName ?sahipIsmi . 
    ?sahipURI pet:hasPhone ?telefon . 
    ?sahipURI pet:hasAddress ?adres . 
} 
ORDER BY ?sahipID 
LIMIT 10


Query 2: Isolate Sub-Class Pet Specifications (Maps to CQ7)

PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> 
PREFIX owl: <http://www.w3.org/2002/07/owl#> 
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#> PREFIX xsd: <http://www.w3.org/2001/XMLSchema#> 
SELECT ?petID ?hayvanTuru ?isim ?irk ?yasYil ?kiloKg 
WHERE { 
    ?evcilHayvanURI pet:hasPetID ?petID . 
    ?evcilHayvanURI rdf:type ?hayvanTuru . 
    ?evcilHayvanURI pet:hasName ?isim . 
    ?evcilHayvanURI pet:hasBreed ?irk . 
    ?evcilHayvanURI pet:hasAge ?yasYil . 
    ?evcilHayvanURI pet:hasWeight ?kiloKg .

ORDER BY ?petID 
LIMIT 10

Query 8: Financial Performance Analytics per Treatment Subclass (Maps to CQ8)

Query 9: Cross-Tabular Animal and Treatment Frequency Analysis (Analytical) 

PREFIX pet: <http://www.semanticweb.org/pet-management-ontology#> 
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

WHERE { 
    ?tedaviURI pet:treatmentType ?tedaviAdi . 
    ?tedaviURI pet:isAppliedTo ?hayvanURI .

    ?tedaviURI rdf:type pet:Surgery .
